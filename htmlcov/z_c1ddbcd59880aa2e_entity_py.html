15from functools import total_ordering 

16 

17 

18@total_ordering 

19class Entity: 

20 """Represents an Entity in an energy system graph. 

21 

22 Abstract superclass of the general types of entities of an energy system 

23 graph, collecting attributes and operations common to all types of nodes. 

24 Users should neither instantiate nor subclass this, but use 

25 :class:`Component`, :class:`Bus`, :class:`Edge` or one of their subclasses 

26 instead. 

27 

28 .. role:: python(code) 

29 :language: python 

30 

31 Parameters 

32 ---------- 

33 label: `hashable`, optional 

34 Used as the string representation of this node. If this parameter is 

35 not an instance of :class:`str` it will be converted to a string and 

36 the result will be used as this node's :attr:`label`, which should be 

37 unique with respect to the other nodes in the energy system graph this 

38 node belongs to. If this parameter is not supplied, the string 

39 representation of this node will instead be generated based on this 

40 nodes `class` and `id`. 

41 

42 custom_properties: `dict` 

43 This dictionary that can be used to store information that can be used 

44 to easily attach custom information to any Entity. 

45 """ 

46 

47 def __init__(self, label, *, custom_properties=None): 

48 self._label = label 

49 if custom_properties is None: 

50 custom_properties = {} 

51 self.custom_properties = custom_properties 

52 

53 def __eq__(self, other): 

54 return str(self) == str(other) 

55 

56 def __lt__(self, other): 

57 return str(self) < str(other) 

58 

59 def __hash__(self): 

60 return hash(str(self)) 

61 

62 def __str__(self): 

63 return str(self.label) 

64 

65 def __repr__(self): 

66 return repr( 

67 "<{0.__module__}.{0.__name__}: {1!r}>".format( 

68 type(self), self.label 

69 ) 

70 ) 

71 

72 @property 

73 def label(self): 

74 """ 

75 If this node was given a `label` on construction, this 

76 attribute holds the actual object passed as a parameter. Otherwise 

77 `node.label` is a synonym for `str(node)`. 

78 """ 

79 try: 

80 return self._label if self._label is not None else self._id_label 

81 except AttributeError: # Workaround for problems with pickle/dill 

82 return hash(self._id_label) 

83 

84 @property 

85 def _id_label(self): 

86 return "<{} #0x{:x}>".format(type(self).__name__, id(self))
